use anchor_lang::prelude::*;
use anchor_spl::token::{self, Mint, MintTo, Token, TokenAccount, Transfer};
use anchor_spl::associated_token::AssociatedToken;

declare_id!("REEMPLAZA_CON_TU_PROGRAM_ID");

#[program]
pub mod lottery {
    use super::*;

    pub fn initialize(ctx: Context<Initialize>) -> Result<()> {
        let global = &mut ctx.accounts.global;
        global.admin = ctx.accounts.payer.key();
        global.round_count = 0;

        let seeds = &[b"mint_authority".as_ref(), &[ctx.bumps.mint_authority]];
        let signer = &[&seeds[..]];
        let cpi = CpiContext::new_with_signer(
            ctx.accounts.token_program.to_account_info(),
            token::InitializeMint {
                mint: ctx.accounts.lottosol_mint.to_account_info(),
                rent: ctx.accounts.rent.to_account_info(),
            },
            signer,
        );
        token::initialize_mint(cpi, 9, ctx.accounts.mint_authority.key, None)?;

        global.lottosol_mint = ctx.accounts.lottosol_mint.key();
        global.treasury = ctx.accounts.treasury.key();
        Ok(())
    }

    pub fn create_round(
        ctx: Context<CreateRound>,
        ticket_price: u64,
        start_time: i64,
        end_time: i64,
        max_tickets: u64,
    ) -> Result<()> {
        let global = &mut ctx.accounts.global;
        require_keys_eq!(global.admin, ctx.accounts.admin.key(), LotteryError::Unauthorized);

        let round = &mut ctx.accounts.round;
        round.id = global.round_count;
        round.ticket_price = ticket_price;
        round.start_time = start_time;
        round.end_time = end_time;
        round.max_tickets = max_tickets;
        round.player_count = 0;
        round.winner = Pubkey::default();
        round.is_finalized = false;
        round.total_prize = 0;
        round.admin = ctx.accounts.admin.key();
        round.bump = ctx.bumps.round;
        round.staking_pool = 0;
        round.total_staked = 0;
        round.staking_finalized = false;

        global.round_count += 1;
        Ok(())
    }

    pub fn buy_ticket(ctx: Context<BuyTicket>, round_id: u64) -> Result<()> {
        let round = &mut ctx.accounts.round;
        let clock = Clock::get()?;

        require!(!round.is_finalized, LotteryError::RoundAlreadyFinalized);
        require!(
            clock.unix_timestamp >= round.start_time && clock.unix_timestamp <= round.end_time,
            LotteryError::RoundNotActive
        );
        require!(round.player_count < round.max_tickets, LotteryError::RoundFull);

        // Transfer SOL a la cuenta de la ronda
        let cpi = CpiContext::new(
            ctx.accounts.system_program.to_account_info(),
            anchor_lang::system_program::Transfer {
                from: ctx.accounts.buyer.to_account_info(),
                to: round.to_account_info(),
            },
        );
        anchor_lang::system_program::transfer(cpi, round.ticket_price)?;

        round.player_count += 1;
        round.total_prize += round.ticket_price;

        // Mintear LottoSol 1:1
        let seeds = &[b"mint_authority".as_ref(), &[ctx.bumps.mint_authority]];
        let signer = &[&seeds[..]];
        let mint_cpi = CpiContext::new_with_signer(
            ctx.accounts.token_program.to_account_info(),
            MintTo {
                mint: ctx.accounts.lottosol_mint.to_account_info(),
                to: ctx.accounts.buyer_lottosol_ata.to_account_info(),
                authority: ctx.accounts.mint_authority.to_account_info(),
            },
            signer,
        );
        token::mint_to(mint_cpi, round.ticket_price)?;

        // Registrar el ticket
        let ticket = &mut ctx.accounts.ticket;
        ticket.owner = ctx.accounts.buyer.key();
        ticket.round_id = round_id;
        ticket.ticket_number = round.player_count; // ya se incrementó

        Ok(())
    }

    pub fn stake(ctx: Context<Stake>, round_id: u64, amount: u64) -> Result<()> {
        let round = &mut ctx.accounts.round;
        let clock = Clock::get()?;

        require!(!round.staking_finalized, LotteryError::StakingAlreadyFinalized);
        require!(clock.unix_timestamp <= round.end_time, LotteryError::RoundEnded);

        // Transferir tokens al vault
        let cpi = CpiContext::new(
            ctx.accounts.token_program.to_account_info(),
            Transfer {
                from: ctx.accounts.user_lottosol_ata.to_account_info(),
                to: ctx.accounts.staking_vault.to_account_info(),
                authority: ctx.accounts.user.to_account_info(),
            },
        );
        token::transfer(cpi, amount)?;

        // Guardar registro de stake
        let stake_entry = &mut ctx.accounts.stake_entry;
        stake_entry.user = ctx.accounts.user.key();
        stake_entry.round_id = round_id;
        stake_entry.amount = amount;
        stake_entry.claimed = false;
        stake_entry.bump = ctx.bumps.stake_entry;

        round.total_staked += amount;

        Ok(())
    }

    pub fn finalize_round(ctx: Context<FinalizeRound>, _round_id: u64) -> Result<()> {
        let winner_index: u64;
        let winner_share: u64;
        let treasury_share: u64;
        let staking_pool: u64;

        {
            let round = &mut ctx.accounts.round;
            require_keys_eq!(round.admin, ctx.accounts.admin.key(), LotteryError::Unauthorized);
            let clock = Clock::get()?;
            require!(!round.is_finalized, LotteryError::RoundAlreadyFinalized);
            require!(clock.unix_timestamp >= round.end_time, LotteryError::RoundStillActive);
            require!(round.player_count > 0, LotteryError::NoTickets);

            winner_index = (clock.slot % round.player_count) + 1;

            // Validar que el ticket proporcionado es el ganador
            let (expected_ticket_pda, _) = Pubkey::find_program_address(
                &[
                    b"ticket",
                    &_round_id.to_le_bytes(),
                    &winner_index.to_le_bytes(),
                ],
                ctx.program_id,
            );
            require_keys_eq!(
                ctx.accounts.ticket.key(),
                expected_ticket_pda,
                LotteryError::WrongWinnerAccount
            );
            require!(
                ctx.accounts.ticket.owner == ctx.accounts.winner.key(),
                LotteryError::WrongWinnerAccount
            );
            require!(
                ctx.accounts.ticket.round_id == _round_id,
                LotteryError::WrongWinnerAccount
            );

            round.winner = ctx.accounts.ticket.owner;
            round.is_finalized = true;

            let total = round.total_prize;
            winner_share = total * 90 / 100;
            treasury_share = total * 5 / 100;
            staking_pool = total - winner_share - treasury_share;

            round.total_prize = 0;                 // evitar doble gasto
            round.staking_pool = staking_pool;
            round.staking_finalized = true;
        }

        // Enviar SOL al ganador
        ctx.accounts.round.sub_lamports(winner_share)?;
        **ctx.accounts.winner.to_account_info().try_borrow_mut_lamports()? += winner_share;

        // Enviar SOL a la tesorería
        ctx.accounts.round.sub_lamports(treasury_share)?;
        **ctx.accounts.treasury.to_account_info().try_borrow_mut_lamports()? += treasury_share;

        Ok(())
    }

    pub fn claim_staking_reward(ctx: Context<ClaimStakingReward>, _round_id: u64) -> Result<()> {
        let round = &ctx.accounts.round;
        require!(round.staking_finalized, LotteryError::StakingNotFinalized);
        require!(!ctx.accounts.stake_entry.claimed, LotteryError::AlreadyClaimed);

        let user_stake = ctx.accounts.stake_entry.amount;
        let total_staked = round.total_staked;
        require!(total_staked > 0, LotteryError::NoStakers);

        let reward = (user_stake as u128)
            .checked_mul(round.staking_pool as u128)
            .unwrap()
            .checked_div(total_staked as u128)
            .unwrap() as u64;

        require!(reward > 0, LotteryError::NoReward);

        ctx.accounts.stake_entry.claimed = true;

        ctx.accounts.round.sub_lamports(reward)?;
        **ctx.accounts.user.to_account_info().try_borrow_mut_lamports()? += reward;

        Ok(())
    }

    pub fn claim_prize(ctx: Context<ClaimPrize>, _round_id: u64) -> Result<()> {
        let round = &mut ctx.accounts.round;
        require!(round.is_finalized, LotteryError::RoundNotFinalized);
        require_keys_eq!(round.winner, ctx.accounts.winner.key(), LotteryError::NotWinner);
        require!(round.total_prize > 0, LotteryError::NoPrize);

        let prize = round.total_prize;
        round.total_prize = 0;

        round.sub_lamports(prize)?;
        **ctx.accounts.winner.to_account_info().try_borrow_mut_lamports()? += prize;

        Ok(())
    }

    pub fn emergency_finalize(
        ctx: Context<EmergencyFinalize>,
        _round_id: u64,
        winner_override: Pubkey,
    ) -> Result<()> {
        let round = &mut ctx.accounts.round;
        require_keys_eq!(round.admin, ctx.accounts.admin.key(), LotteryError::Unauthorized);
        require!(!round.is_finalized, LotteryError::RoundAlreadyFinalized);

        round.winner = winner_override;
        round.is_finalized = true;

        let total = round.total_prize;
        let winner_share = total * 90 / 100;
        let treasury_share = total * 5 / 100;
        let staking_pool = total - winner_share - treasury_share;

        round.total_prize = 0;                 // evitar doble gasto
        round.staking_pool = staking_pool;
        round.staking_finalized = true;

        // Transferir al ganador forzado
        ctx.accounts.round.sub_lamports(winner_share)?;
        **ctx.accounts.winner_override.to_account_info().try_borrow_mut_lamports()? += winner_share;

        // Tesorería
        ctx.accounts.round.sub_lamports(treasury_share)?;
        **ctx.accounts.treasury.to_account_info().try_borrow_mut_lamports()? += treasury_share;

        Ok(())
    }
}

// ──── Contextos (Accounts) ────

#[derive(Accounts)]
pub struct Initialize<'info> {
    #[account(
        init,
        payer = payer,
        space = 8 + std::mem::size_of::<LotteryGlobal>(),
        seeds = [b"global"],
        bump
    )]
    pub global: Account<'info, LotteryGlobal>,

    #[account(mut)]
    pub payer: Signer<'info>,

    /// CHECK: PDA autoridad de minteo.
    #[account(
        seeds = [b"mint_authority"],
        bump
    )]
    pub mint_authority: AccountInfo<'info>,

    #[account(
        init,
        payer = payer,
        mint::decimals = 9,
        mint::authority = mint_authority.key(),
    )]
    pub lottosol_mint: Account<'info, Mint>,

    /// CHECK: Tesorería elegida por el admin.
    pub treasury: AccountInfo<'info>,

    pub system_program: Program<'info, System>,
    pub token_program: Program<'info, Token>,
    pub rent: Sysvar<'info, Rent>,
}

#[derive(Accounts)]
pub struct CreateRound<'info> {
    #[account(mut, seeds = [b"global"], bump)]
    pub global: Account<'info, LotteryGlobal>,

    #[account(
        init,
        payer = admin,
        space = 8 + std::mem::size_of::<Round>(),
        seeds = [b"round", global.round_count.to_le_bytes().as_ref()],
        bump
    )]
    pub round: Account<'info, Round>,

    #[account(mut)]
    pub admin: Signer<'info>,
    pub system_program: Program<'info, System>,
}

#[derive(Accounts)]
#[instruction(round_id: u64)]
pub struct BuyTicket<'info> {
    #[account(
        mut,
        seeds = [b"round", round_id.to_le_bytes().as_ref()],
        bump = round.bump
    )]
    pub round: Account<'info, Round>,

    #[account(
        init,
        payer = buyer,
        space = 8 + std::mem::size_of::<Ticket>(),
        seeds = [
            b"ticket",
            round_id.to_le_bytes().as_ref(),
            (round.player_count + 1).to_le_bytes().as_ref()
        ],
        bump
    )]
    pub ticket: Account<'info, Ticket>,

    #[account(mut)]
    pub buyer: Signer<'info>,

    #[account(
        mut,
        address = global.lottosol_mint   // ✅ valida que el mint sea el oficial
    )]
    pub lottosol_mint: Account<'info, Mint>,

    #[account(
        seeds = [b"global"],
        bump
    )]
    pub global: Account<'info, LotteryGlobal>, // ✅ añadido

    /// CHECK: PDA que firma el mint_to.
    #[account(
        seeds = [b"mint_authority"],
        bump
    )]
    pub mint_authority: AccountInfo<'info>,

    #[account(
        init_if_needed,
        payer = buyer,
        associated_token::mint = lottosol_mint,
        associated_token::authority = buyer,
    )]
    pub buyer_lottosol_ata: Account<'info, TokenAccount>,

    pub system_program: Program<'info, System>,
    pub token_program: Program<'info, Token>,
    pub associated_token_program: Program<'info, AssociatedToken>,
    pub rent: Sysvar<'info, Rent>,
}

// ──── Estructuras de datos ────

#[account]
pub struct LotteryGlobal {
    pub admin: Pubkey,
    pub round_count: u64,
    pub lottosol_mint: Pubkey,
    pub treasury: Pubkey,
}

#[account]
pub struct Round {
    pub id: u64,
    pub ticket_price: u64,
    pub start_time: i64,
    pub end_time: i64,
    pub max_tickets: u64,
    pub player_count: u64,
    pub winner: Pubkey,
    pub is_finalized: bool,
    pub total_prize: u64,
    pub admin: Pubkey,
    pub bump: u8,
    pub staking_pool: u64,
    pub total_staked: u64,
    pub staking_finalized: bool,
}

#[account]
pub struct Ticket {
    pub owner: Pubkey,
    pub round_id: u64,
    pub ticket_number: u64,
}

#[account]
pub struct StakeEntry {
    pub user: Pubkey,
    pub round_id: u64,
    pub amount: u64,
    pub claimed: bool,
    pub bump: u8,
}

// ──── Errores ────

#[error_code]
pub enum LotteryError {
    #[msg("No autorizado")]
    Unauthorized,
    #[msg("Ronda ya finalizada")]
    RoundAlreadyFinalized,
    #[msg("Ronda no activa")]
    RoundNotActive,
    #[msg("Ronda llena")]
    RoundFull,
    #[msg("Ronda aún activa")]
    RoundStillActive,
    #[msg("No hay tickets")]
    NoTickets,
    #[msg("No finalizada")]
    RoundNotFinalized,
    #[msg("No eres el ganador")]
    NotWinner,
    #[msg("Sin premio")]
    NoPrize,
    #[msg("Winner incorrecto")]
    WrongWinnerAccount,
    #[msg("Staking ya finalizado")]
    StakingAlreadyFinalized,
    #[msg("La ronda ya terminó")]
    RoundEnded,
    #[msg("Staking no finalizado")]
    StakingNotFinalized,
    #[msg("Ya reclamaste tu recompensa")]
    AlreadyClaimed,
    #[msg("No hay stakers")]
    NoStakers,
    #[msg("Sin recompensa")]
    NoReward,
}
