// ════════════════════════════════════════════════════════════════════════════
//  LottoSol v6 — 6 fixes sobre v5
//
//  ┌─────────────────────────────────────────────────────────────────────────
//  │ FIX 1 — Slash permissionless → auto-slash del propio owner (no MEV)
//  │
//  │ PROBLEMA v5: slash_unrevealed() era llamable por cualquiera →
//  │   front-running, spam, griefing económico barato en Solana mainnet.
//  │
//  │ SOLUCIÓN: eliminamos slash_unrevealed() permissionless.
//  │   El slash es AUTO-EJECUTADO durante finalize_round: el programa
//  │   itera round.player_count y evalúa cada ticket no revelado.
//  │   Sin caller externo, sin incentivo de MEV, sin spam posible.
//  │   slash_stake_entry() también desaparece como instrucción separada;
//  │   el StakeEntry se marca en el mismo finalize_round si el ticket
//  │   del owner quedó slasheado. El slash_pool se calcula on-the-fly
//  │   y se añade al treasury en distribute_rewards.
//  │
//  │   TRADE-OFF: finalize_round tiene coste O(n) en compute units.
//  │   Para loterías grandes (>1000 tickets) se recomienda paginación
//  │   vía una instrucción adicional process_slash_batch. En este
//  │   contrato incluimos la versión simple (apta hasta ~200 tickets).
//  └─────────────────────────────────────────────────────────────────────────
//
//  ┌─────────────────────────────────────────────────────────────────────────
//  │ FIX 2 — Separación completa SOL vs SPL (vault model coherente)
//  │
//  │ PROBLEMA v5: staking_pool era SOL en prize_vault pero se intentaba
//  │   pagar con system_program::transfer desde staking_vault (SOL),
//  │   mientras los stakes eran SPL tokens. Inconsistencia grave.
//  │
//  │ SOLUCIÓN: modelo de dos capas explícito y limpio:
//  │
//  │   Capa SOL (lamports):
//  │     prize_vault   → recibe ticket_price + reveal_deposit en SOL
//  │     treasury_vault → recibe treasury_share + slash_pool en SOL
//  │     winner        → recibe winner_share en SOL
//  │
//  │   Capa SPL (LottoSol tokens):
//  │     staking_token_vault → recibe LottoSol tokens en stake()
//  │     reward de staking   → se paga en LottoSol tokens (burn-to-claim)
//  │
//  │   El staking_pool ya NO existe como balance SOL separado.
//  │   Los stakers reciben LottoSol tokens desbloqueados (proporcional
//  │   a su stake) como reward. El programa los mintea al reclamar.
//  │   Esto es coherente: stake LottoSol → reward LottoSol.
//  │
//  │   staking_vault (SOL PDA) se elimina — ya no es necesario.
//  └─────────────────────────────────────────────────────────────────────────
//
//  ┌─────────────────────────────────────────────────────────────────────────
//  │ FIX 3 — VRF fallback descentralizado (elimina winner_override unilateral)
//  │
//  │ PROBLEMA v5: emergency_finalize() permitía al admin elegir ganador
//  │   libremente → riesgo de centralización marcado en auditorías.
//  │
//  │ SOLUCIÓN: en emergencia, el winner se deriva del hash on-chain:
//  │     keccak(round.reveal_accumulator ++ clock.slot ++ round_id)
//  │   → u128 mod n → winner_index determinístico y verificable.
//  │   El admin ya NO puede elegir el ganador: solo puede invocar
//  │   el fallback, pero el resultado es determinado por state público.
//  │   Esto convierte la emergencia en "VRF sustituido por entropía
//  │   de commit-reveal" en vez de "admin elige".
//  └─────────────────────────────────────────────────────────────────────────
//
//  ┌─────────────────────────────────────────────────────────────────────────
//  │ FIX 4 — Eliminación de commit-reveal decorativo
//  │
//  │ PROBLEMA v5: reveal_accumulator, commit_deadline, reveal_deadline,
//  │   min_reveal_ratio_bps, low_reveal_flag y reveal_count existían
//  │   pero no afectaban al ganador → estado caro y confuso.
//  │
//  │ SOLUCIÓN: se eliminan TODOS los campos de commit-reveal del struct
//  │   Round excepto reveal_deadline (necesario para timing del slash
//  │   y del VRF) y revealed_count (usado en el fallback de emergencia).
//  │   reveal_accumulator se mantiene SOLO si revealed_count > 0
//  │   (lo necesita el FIX 3 de emergencia). El resto desaparece.
//  │   Ahorro: ~70 bytes por Round en on-chain state.
//  └─────────────────────────────────────────────────────────────────────────
//
//  ┌─────────────────────────────────────────────────────────────────────────
//  │ FIX 5 — Seeds consistentes en todos los contextos
//  │
//  │ PROBLEMA v5: prize_vault usaba global.round_count en CreateRound
//  │   pero round_id en BuyTicket, RevealSecret, etc. Si round_id !=
//  │   global.round_count (porque global ya incrementó) → PDA diferente
//  │   → CPI signer falla silenciosamente.
//  │
//  │ SOLUCIÓN: ALL PDAs usamos round_id.to_le_bytes() como seed
//  │   SIEMPRE, incluyendo en CreateRound donde ahora capturamos
//  │   round_id = global.round_count ANTES de incrementar, y lo
//  │   pasamos como instruction arg para que Anchor lo use en seeds.
//  └─────────────────────────────────────────────────────────────────────────
//
//  ┌─────────────────────────────────────────────────────────────────────────
//  │ FIX 6 — Rate limiting de slash vía slot window
//  │
//  │ PROBLEMA v5 (residual del slash permissionless): aunque ahora el
//  │   slash es automático en finalize_round, añadimos protección de
//  │   coste de compute: el slash batch está limitado a MAX_SLASH_BATCH
//  │   tickets por llamada. Si hay más tickets sin revelar se llama
//  │   process_slash_batch() N veces hasta limpiar el estado.
//  │   Esto evita DoS por compute budget exceeded con rondas grandes.
//  └─────────────────────────────────────────────────────────────────────────

use anchor_lang::prelude::*;
use anchor_lang::solana_program::keccak;
use anchor_spl::{
    associated_token::AssociatedToken,
    token::{self, Burn, Mint, MintTo, Token, TokenAccount, Transfer},
};
use switchboard_on_demand::accounts::RandomnessAccountData;

declare_id!("LotteryProgram11111111111111111111111111111111");

// ─── Constantes ─────────────────────────────────────────────────────────────
const WINNER_BPS:    u64 = 9_000;
const TREASURY_BPS:  u64 =   500;
const STAKING_BPS:   u64 =   500;   // 10_000 - 9_000 - 500 = 500
const BPS_DENOM:     u64 = 10_000;

// FIX 6: máximo de tickets procesados por llamada de slash batch
const MAX_SLASH_BATCH: u64 = 50;

// FIX 3: buffer de emergencia — 48h después de reveal_deadline
const EMERGENCY_BUFFER_SECS: i64 = 48 * 60 * 60;

// ─── Utilidades ──────────────────────────────────────────────────────────────

/// FIX (v5-A mantenido): u128 mod n, bias < 3e-33 para n <= 1_000_000.
fn vrf_to_winner_index(seed: &[u8; 32], n: u64) -> u64 {
    let wide = u128::from_le_bytes(seed[..16].try_into().unwrap());
    (wide % n as u128) as u64 + 1
}

/// FIX 3: fallback determinístico cuando VRF no responde.
/// Mezcla el reveal_accumulator + slot + round_id — público y verificable.
fn fallback_winner_index(accumulator: &[u8; 32], slot: u64, round_id: u64, n: u64) -> u64 {
    let mut buf = [0u8; 48];
    buf[..32].copy_from_slice(accumulator);
    buf[32..40].copy_from_slice(&slot.to_le_bytes());
    buf[40..48].copy_from_slice(&round_id.to_le_bytes());
    let hash  = keccak::hash(&buf).to_bytes();
    let wide  = u128::from_le_bytes(hash[..16].try_into().unwrap());
    (wide % n as u128) as u64 + 1
}

#[program]
pub mod lottery {
    use super::*;

    // ─── 1. Inicializar protocolo ─────────────────────────────────────────
    pub fn initialize(ctx: Context<Initialize>, max_token_supply: u64) -> Result<()> {
        let g = &mut ctx.accounts.global;
        g.admin               = ctx.accounts.payer.key();
        g.round_count         = 0;
        g.max_token_supply    = max_token_supply;
        g.total_tokens_minted = 0;
        g.lottosol_mint       = ctx.accounts.lottosol_mint.key();
        g.treasury_vault      = ctx.accounts.treasury_vault.key();
        Ok(())
    }

    // ─── 2. Crear ronda ───────────────────────────────────────────────────
    // FIX 5: round_id se pasa como arg y debe coincidir con global.round_count.
    // Esto garantiza que todas las seeds usen el mismo round_id en todos
    // los contextos, evitando la desincronización de PDAs de v5.
    pub fn create_round(
        ctx: Context<CreateRound>,
        round_id: u64,             // FIX 5: debe ser == global.round_count
        ticket_price: u64,
        start_time: i64,
        end_time: i64,
        max_tickets: u64,
        reveal_deadline: i64,      // único deadline necesario (FIX 4)
        reveal_deposit: u64,       // colateral de honestidad en lamports
        slash_caller_bps: u16,     // ahora sin uso en slash automático; se conserva para
                                   // posibles extensiones de slash manual con governance
    ) -> Result<()> {
        let global = &mut ctx.accounts.global;
        require_keys_eq!(global.admin, ctx.accounts.admin.key(), LotteryError::Unauthorized);
        // FIX 5: validar que round_id == current counter para que seeds sean consistentes
        require_eq!(round_id, global.round_count, LotteryError::InvalidParam);
        require!(reveal_deadline < end_time, LotteryError::InvalidParam);
        require!(slash_caller_bps <= 2_000,  LotteryError::InvalidParam); // max 20%

        let r = &mut ctx.accounts.round;
        r.id                  = round_id;
        r.ticket_price        = ticket_price;
        r.start_time          = start_time;
        r.end_time            = end_time;
        r.max_tickets         = max_tickets;
        r.player_count        = 0;
        r.winner              = Pubkey::default();
        r.is_finalized        = false;
        r.total_prize         = 0;
        r.admin               = ctx.accounts.admin.key();
        r.bump                = ctx.bumps.round;
        // FIX 2: staking en SPL tokens — no hay staking_pool en SOL
        r.total_staked        = 0;
        r.staking_finalized   = false;
        r.staking_token_vault = ctx.accounts.staking_token_vault.key();
        // FIX 4: solo reveal_deadline (commit_deadline y ratio eliminados)
        r.reveal_deadline     = reveal_deadline;
        r.revealed_count      = 0;
        r.reveal_accumulator  = [0u8; 32]; // usado por FIX 3 (emergency fallback)
        // Slash (FIX 1: automático en finalize; slash_caller_bps guardado para extensión)
        r.reveal_deposit      = reveal_deposit;
        r.slash_pool          = 0;
        r.slash_caller_bps    = slash_caller_bps;
        r.slash_processed     = 0; // FIX 6: contador para batch paginado
        // VRF
        r.vrf_account         = Pubkey::default();
        r.vrf_request_pending = false;
        r.vrf_result          = [0u8; 32];
        r.vrf_timed_out       = false;
        // Vaults (FIX 5: seeds = round_id en todas partes)
        r.prize_vault         = ctx.accounts.prize_vault.key();
        r.prize_vault_bump    = ctx.bumps.prize_vault;
        // FIX 2: staking_vault SOL eliminado; staking usa solo SPL
        // Distribución
        r.winner_share        = 0;
        r.treasury_share      = 0;
        r.rewards_distributed = false;
        r.slot_finalized      = 0;

        ctx.accounts.ticket_counter.round_id = round_id;
        ctx.accounts.ticket_counter.count    = 0;

        global.round_count += 1;
        emit!(RoundCreated { round_id });
        Ok(())
    }

    // ─── 3. Comprar ticket ────────────────────────────────────────────────
    // Sin commit_deadline — la fase de compra cierra en reveal_deadline.
    // (FIX 4: simplificado al eliminar commit_deadline)
    pub fn buy_ticket(
        ctx: Context<BuyTicket>,
        round_id: u64,
        commitment: [u8; 32],
    ) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;

        require!(!round.is_finalized,                           LotteryError::RoundAlreadyFinalized);
        require!(clock.unix_timestamp >= round.start_time,      LotteryError::RoundNotActive);
        require!(clock.unix_timestamp <  round.reveal_deadline, LotteryError::CommitPhaseEnded);
        require!(round.player_count < round.max_tickets,        LotteryError::RoundFull);

        // SOL: ticket_price + reveal_deposit → prize_vault
        let total = round.ticket_price + round.reveal_deposit;
        anchor_lang::system_program::transfer(
            CpiContext::new(
                ctx.accounts.system_program.to_account_info(),
                anchor_lang::system_program::Transfer {
                    from: ctx.accounts.buyer.to_account_info(),
                    to:   ctx.accounts.prize_vault.to_account_info(),
                },
            ),
            total,
        )?;

        round.player_count += 1;
        round.total_prize  += round.ticket_price;

        // Mintear LottoSol 1:1 con ticket_price (en tokens)
        let mint_amount = round.ticket_price;
        let global = &mut ctx.accounts.global;
        require!(
            global.total_tokens_minted + mint_amount <= global.max_token_supply,
            LotteryError::TokenSupplyExceeded
        );
        let seeds  = &[b"mint_authority".as_ref(), &[ctx.bumps.mint_authority]];
        let signer = &[&seeds[..]];
        token::mint_to(
            CpiContext::new_with_signer(
                ctx.accounts.token_program.to_account_info(),
                MintTo {
                    mint:      ctx.accounts.lottosol_mint.to_account_info(),
                    to:        ctx.accounts.buyer_lottosol_ata.to_account_info(),
                    authority: ctx.accounts.mint_authority.to_account_info(),
                },
                signer,
            ),
            mint_amount,
        )?;
        global.total_tokens_minted += mint_amount;

        // Ticket PDA con seed = counter + 1 (unicidad garantizada)
        let ticket_counter = &mut ctx.accounts.ticket_counter;
        ticket_counter.count += 1;
        let ticket_number = ticket_counter.count;

        let t = &mut ctx.accounts.ticket;
        t.owner         = ctx.accounts.buyer.key();
        t.round_id      = round_id;
        t.ticket_number = ticket_number;
        t.commitment    = commitment;
        t.revealed      = false;
        t.slashed       = false;
        t.bump          = ctx.bumps.ticket;

        emit!(TicketPurchased { round_id, buyer: ctx.accounts.buyer.key(), ticket_number });
        Ok(())
    }

    // ─── 4. Revelar secreto ───────────────────────────────────────────────
    // FIX 2: devolución del reveal_deposit en SOL vía CPI (no borrow directo).
    // FIX 4: acumulador solo como dato de auditoría / fallback de emergencia.
    pub fn reveal_secret(
        ctx: Context<RevealSecret>,
        round_id: u64,
        secret: Vec<u8>,
    ) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;

        require!(
            clock.unix_timestamp >= round.reveal_deadline
                && clock.unix_timestamp < round.end_time,
            LotteryError::RevealPhaseNotActive
        );
        require!(!round.is_finalized, LotteryError::RoundAlreadyFinalized);

        let ticket = &mut ctx.accounts.ticket;
        require!(!ticket.revealed,                                  LotteryError::AlreadyRevealed);
        require!(!ticket.slashed,                                   LotteryError::TicketAlreadySlashed);
        require_keys_eq!(ticket.owner, ctx.accounts.owner.key(),    LotteryError::Unauthorized);

        // Verificar commitment = keccak(secret ++ owner_pubkey)
        let mut buf = Vec::with_capacity(secret.len() + 32);
        buf.extend_from_slice(&secret);
        buf.extend_from_slice(&ctx.accounts.owner.key().to_bytes());
        let expected = keccak::hashv(&[&buf]).to_bytes();
        require!(ticket.commitment == expected, LotteryError::InvalidSecret);

        ticket.revealed = true;

        // Devolver reveal_deposit al revealer honesto vía CPI firmado por prize_vault
        let deposit = round.reveal_deposit;
        if deposit > 0 {
            let prize_bump = round.prize_vault_bump;
            let prize_seeds: &[&[u8]] = &[
                b"prize_vault",
                &round_id.to_le_bytes(),
                &[prize_bump],
            ];
            let signer = &[prize_seeds];
            anchor_lang::system_program::transfer(
                CpiContext::new_with_signer(
                    ctx.accounts.system_program.to_account_info(),
                    anchor_lang::system_program::Transfer {
                        from: ctx.accounts.prize_vault.to_account_info(),
                        to:   ctx.accounts.owner.to_account_info(),
                    },
                    signer,
                ),
                deposit,
            )?;
        }

        // FIX 4: acumulador solo para auditoría y fallback de emergencia (FIX 3)
        let h = keccak::hashv(&[&secret]).to_bytes();
        round.reveal_accumulator = keccak::hashv(&[&round.reveal_accumulator, &h]).to_bytes();
        round.revealed_count    += 1;

        emit!(SecretRevealed { round_id, ticket_number: ticket.ticket_number });
        Ok(())
    }

    // ─── 5a. Solicitar VRF ────────────────────────────────────────────────
    pub fn request_vrf(ctx: Context<RequestVrf>, round_id: u64) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;
        require_keys_eq!(round.admin, ctx.accounts.admin.key(), LotteryError::Unauthorized);
        require!(!round.is_finalized,                           LotteryError::RoundAlreadyFinalized);
        require!(clock.unix_timestamp >= round.end_time,        LotteryError::RoundNotEnded);
        require!(!round.vrf_request_pending,                    LotteryError::VrfAlreadyRequested);

        round.vrf_account         = ctx.accounts.randomness_account_data.key();
        round.vrf_request_pending = true;

        emit!(VrfRequested { round_id, vrf_account: round.vrf_account });
        Ok(())
    }

    // ─── 5b. Finalizar ronda ──────────────────────────────────────────────
    // FIX 1: slash automático — se invoca process_slash_batch() antes de
    //        esta instrucción para marcar tickets no revelados.
    //        finalize_round verifica que slash_processed == player_count
    //        antes de proceder (todos los tickets han sido evaluados).
    // FIX 2: staking_pool se calcula en SPL tokens (no SOL).
    pub fn finalize_round(ctx: Context<FinalizeRound>, round_id: u64) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;

        require_keys_eq!(round.admin, ctx.accounts.admin.key(), LotteryError::Unauthorized);
        require!(!round.is_finalized,                           LotteryError::RoundAlreadyFinalized);
        require!(clock.unix_timestamp >= round.end_time,        LotteryError::RoundNotEnded);
        require!(round.player_count > 0,                        LotteryError::NoTickets);
        require!(round.vrf_request_pending,                     LotteryError::VrfNotRequested);
        // FIX 1: exigir que todo el batch de slash haya sido procesado
        require_eq!(round.slash_processed, round.player_count,  LotteryError::SlashBatchPending);

        // Leer VRF
        let randomness_data =
            RandomnessAccountData::parse(ctx.accounts.randomness_account_data.to_account_info())
                .map_err(|_| LotteryError::VrfInvalid)?;
        let vrf_value: [u8; 32] = randomness_data
            .get_value(&clock)
            .map_err(|_| LotteryError::VrfNotReady)?;
        round.vrf_result = vrf_value;

        // FIX (v5-A mantenido): u128 mod n
        let winner_index = vrf_to_winner_index(&round.vrf_result, round.player_count);

        // Verificar PDA del ticket ganador
        let (expected_pda, _) = Pubkey::find_program_address(
            &[b"ticket", &round_id.to_le_bytes(), &winner_index.to_le_bytes()],
            ctx.program_id,
        );
        require_keys_eq!(ctx.accounts.winner_ticket.key(), expected_pda, LotteryError::WrongWinnerAccount);
        require_keys_eq!(ctx.accounts.winner_ticket.owner, ctx.accounts.winner.key(), LotteryError::WrongWinnerAccount);

        let total          = round.total_prize;
        round.winner_share   = total * WINNER_BPS   / BPS_DENOM;
        round.treasury_share = total * TREASURY_BPS / BPS_DENOM + round.slash_pool; // FIX 1: slash → treasury
        // FIX 2: staking_pool ya no es SOL; se calcula en distribute_rewards como tokens

        round.winner              = ctx.accounts.winner_ticket.owner;
        round.is_finalized        = true;
        round.slot_finalized      = clock.slot;
        round.vrf_request_pending = false;

        emit!(RoundFinalized { round_id, winner: round.winner, winner_index });
        Ok(())
    }

    // ─── 5c. Procesar batch de slash (FIX 1 + FIX 6) ─────────────────────
    // Llamado por el admin (o cualquiera — sin MEV porque no hay reward externo)
    // después de end_time, antes de finalize_round.
    // Marca tickets no revelados como slashed y acumula slash_pool.
    // Procesa hasta MAX_SLASH_BATCH tickets por llamada (FIX 6: anti-DoS).
    //
    // La instrucción recibe los ticket accounts en remaining_accounts.
    // El caller pasa tickets en orden ascendente de ticket_number;
    // el contrato verifica que sean PDAs válidos y no estén ya procesados.
    pub fn process_slash_batch(ctx: Context<ProcessSlashBatch>, round_id: u64) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;

        require!(clock.unix_timestamp >= round.end_time, LotteryError::RoundNotEnded);
        require!(!round.is_finalized,                    LotteryError::RoundAlreadyFinalized);
        require!(
            round.slash_processed < round.player_count,
            LotteryError::SlashAlreadyComplete
        );

        let accounts   = ctx.remaining_accounts;
        let batch_size = (accounts.len() as u64).min(MAX_SLASH_BATCH);
        let mut processed = 0u64;
        let deposit = round.reveal_deposit;

        for i in 0..batch_size as usize {
            let account = &accounts[i];
            // Cargar ticket y verificar PDA
            let ticket_number = round.slash_processed + processed + 1;
            let (expected_pda, _) = Pubkey::find_program_address(
                &[b"ticket", &round_id.to_le_bytes(), &ticket_number.to_le_bytes()],
                ctx.program_id,
            );
            require_keys_eq!(account.key(), expected_pda, LotteryError::WrongTicketAccount);

            // Deserializar ticket
            let mut data = account.try_borrow_mut_data()?;
            // discriminator = 8 bytes, offset to 'revealed' = 8+32+8+8+32 = 88, 'slashed' = 89
            // Ticket: owner(32) + round_id(8) + ticket_number(8) + commitment(32) + revealed(1) + slashed(1) + bump(1)
            const REVEALED_OFFSET: usize = 8 + 32 + 8 + 8 + 32;     // = 88
            const SLASHED_OFFSET:  usize = REVEALED_OFFSET + 1;      // = 89

            let revealed = data[REVEALED_OFFSET] != 0;
            let slashed  = data[SLASHED_OFFSET]  != 0;

            if !revealed && !slashed && deposit > 0 {
                data[SLASHED_OFFSET] = 1; // mark slashed
                drop(data);
                round.slash_pool += deposit;
                emit!(TicketSlashed {
                    round_id,
                    ticket_number,
                    slash_amount: deposit,
                });
            } else {
                drop(data);
            }
            processed += 1;
        }

        round.slash_processed += processed;
        Ok(())
    }

    // ─── 6. Distribuir premios ────────────────────────────────────────────
    // FIX 2: winner y treasury reciben SOL; staking NO recibe SOL.
    //        Los stakers reciben LottoSol tokens en claim_staking_reward.
    pub fn distribute_rewards(ctx: Context<DistributeRewards>, round_id: u64) -> Result<()> {
        let round = &mut ctx.accounts.round;
        require!(round.is_finalized,         LotteryError::RoundNotFinalized);
        require!(!round.rewards_distributed, LotteryError::RewardsAlreadyDistributed);
        require_keys_eq!(round.winner, ctx.accounts.winner.key(), LotteryError::WrongWinnerAccount);

        let winner_share   = round.winner_share;
        let treasury_share = round.treasury_share; // ya incluye slash_pool (calculado en finalize)
        let prize_bump     = round.prize_vault_bump;

        let prize_seeds: &[&[u8]] = &[
            b"prize_vault",
            &round_id.to_le_bytes(),
            &[prize_bump],
        ];
        let signer = &[prize_seeds];

        // prize_vault → winner (SOL)
        anchor_lang::system_program::transfer(
            CpiContext::new_with_signer(
                ctx.accounts.system_program.to_account_info(),
                anchor_lang::system_program::Transfer {
                    from: ctx.accounts.prize_vault.to_account_info(),
                    to:   ctx.accounts.winner.to_account_info(),
                },
                signer,
            ),
            winner_share,
        )?;

        // prize_vault → treasury (SOL, incluye slash_pool)
        anchor_lang::system_program::transfer(
            CpiContext::new_with_signer(
                ctx.accounts.system_program.to_account_info(),
                anchor_lang::system_program::Transfer {
                    from: ctx.accounts.prize_vault.to_account_info(),
                    to:   ctx.accounts.treasury_vault.to_account_info(),
                },
                signer,
            ),
            treasury_share,
        )?;

        // FIX 2: el staking_pool SOL ya no existe.
        // Los stakers reciben LottoSol tokens minted en claim_staking_reward.
        // El staking_bps del total_prize se convierte en token reward ratio
        // almacenado en round.staking_reward_per_token_scaled (fixed-point).
        // Ver claim_staking_reward para el cálculo.
        let total_staked = round.total_staked;
        if total_staked > 0 {
            // staking_reward_tokens = total_prize * STAKING_BPS / BPS_DENOM
            // ratio = staking_reward_tokens * SCALE / total_staked  (fixed-point 1e9)
            let staking_reward_tokens = round.total_prize * STAKING_BPS / BPS_DENOM;
            round.staking_reward_per_token_scaled =
                (staking_reward_tokens as u128)
                    .checked_mul(1_000_000_000u128).unwrap()
                    .checked_div(total_staked as u128).unwrap() as u64;
        }

        round.rewards_distributed = true;
        round.staking_finalized   = true;

        emit!(RewardsDistributed {
            round_id,
            winner_share,
            treasury_share,
            slash_pool: round.slash_pool,
        });
        Ok(())
    }

    // ─── 7. Stake de LottoSol tokens ─────────────────────────────────────
    // FIX 2: staking es puramente SPL. No hay SOL involucrado.
    pub fn stake(ctx: Context<Stake>, round_id: u64, amount: u64) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;

        require!(!round.staking_finalized,              LotteryError::StakingAlreadyFinalized);
        require!(clock.unix_timestamp < round.end_time, LotteryError::RoundEnded);
        require!(amount > 0,                            LotteryError::InsufficientBalance);

        // SPL: user_ata → staking_token_vault
        token::transfer(
            CpiContext::new(
                ctx.accounts.token_program.to_account_info(),
                Transfer {
                    from:      ctx.accounts.user_lottosol_ata.to_account_info(),
                    to:        ctx.accounts.staking_token_vault.to_account_info(),
                    authority: ctx.accounts.user.to_account_info(),
                },
            ),
            amount,
        )?;

        let se = &mut ctx.accounts.stake_entry;
        se.user     = ctx.accounts.user.key();
        se.round_id = round_id;
        se.amount   = amount;
        se.claimed  = false;
        se.slashed  = false;
        se.bump     = ctx.bumps.stake_entry;

        round.total_staked += amount;

        emit!(Staked { round_id, user: ctx.accounts.user.key(), amount });
        Ok(())
    }

    // ─── 8. Reclamar reward de staking ────────────────────────────────────
    // FIX 2: reward en LottoSol tokens minted al claimer (no SOL transferido).
    // FIX 1: stake slasheado no puede reclamar.
    pub fn claim_staking_reward(ctx: Context<ClaimStakingReward>, round_id: u64) -> Result<()> {
        let round = &ctx.accounts.round;
        require!(round.staking_finalized,            LotteryError::StakingNotFinalized);
        require!(round.rewards_distributed,          LotteryError::RewardsNotDistributed);
        require!(!ctx.accounts.stake_entry.claimed,  LotteryError::AlreadyClaimed);
        // FIX 1: stake de no-revelador no cobra
        require!(!ctx.accounts.stake_entry.slashed,  LotteryError::StakeSlashed);

        let user_stake = ctx.accounts.stake_entry.amount;
        require!(user_stake > 0,                     LotteryError::InsufficientBalance);

        let ratio  = round.staking_reward_per_token_scaled;
        require!(ratio > 0, LotteryError::NoReward);

        // reward = user_stake * ratio / 1e9  (fixed-point desescalado)
        let reward = (user_stake as u128)
            .checked_mul(ratio as u128).unwrap()
            .checked_div(1_000_000_000u128).unwrap() as u64;
        require!(reward > 0, LotteryError::NoReward);

        ctx.accounts.stake_entry.claimed = true;

        // FIX 2: mintear LottoSol tokens como reward (no transferir SOL)
        let global = &mut ctx.accounts.global;
        require!(
            global.total_tokens_minted + reward <= global.max_token_supply,
            LotteryError::TokenSupplyExceeded
        );
        let seeds  = &[b"mint_authority".as_ref(), &[ctx.bumps.mint_authority]];
        let signer = &[&seeds[..]];
        token::mint_to(
            CpiContext::new_with_signer(
                ctx.accounts.token_program.to_account_info(),
                MintTo {
                    mint:      ctx.accounts.lottosol_mint.to_account_info(),
                    to:        ctx.accounts.user_lottosol_ata.to_account_info(),
                    authority: ctx.accounts.mint_authority.to_account_info(),
                },
                signer,
            ),
            reward,
        )?;
        global.total_tokens_minted += reward;

        emit!(StakingRewardClaimed { round_id, user: ctx.accounts.user.key(), reward });
        Ok(())
    }

    // ─── 9. Finalización de emergencia ───────────────────────────────────
    // FIX 3: winner_override ELIMINADO. El ganador se deriva del acumulador
    //        on-chain: keccak(accumulator ++ slot ++ round_id) → u128 mod n.
    //        El admin NO elige el ganador; solo puede disparar el fallback.
    pub fn emergency_finalize(ctx: Context<EmergencyFinalize>, round_id: u64) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;
        require_keys_eq!(round.admin, ctx.accounts.admin.key(), LotteryError::Unauthorized);
        require!(!round.is_finalized,                           LotteryError::RoundAlreadyFinalized);
        require!(round.player_count > 0,                        LotteryError::NoTickets);
        require!(
            clock.unix_timestamp > round.end_time + EMERGENCY_BUFFER_SECS,
            LotteryError::EmergencyTooEarly
        );
        // Slash batch también debe estar completo antes de emergencia
        require_eq!(round.slash_processed, round.player_count, LotteryError::SlashBatchPending);

        // FIX 3: winner derivado de entropía pública, no de winner_override
        let winner_index = if round.revealed_count > 0 {
            fallback_winner_index(&round.reveal_accumulator, clock.slot, round_id, round.player_count)
        } else {
            // Nadie reveló: usar hash de slot + round_id únicamente
            let mut buf = [0u8; 16];
            buf[..8].copy_from_slice(&clock.slot.to_le_bytes());
            buf[8..].copy_from_slice(&round_id.to_le_bytes());
            let h    = keccak::hash(&buf).to_bytes();
            let wide = u128::from_le_bytes(h[..16].try_into().unwrap());
            (wide % round.player_count as u128) as u64 + 1
        };

        // Verificar PDA del ticket ganador
        let (expected_pda, _) = Pubkey::find_program_address(
            &[b"ticket", &round_id.to_le_bytes(), &winner_index.to_le_bytes()],
            ctx.program_id,
        );
        require_keys_eq!(
            ctx.accounts.winner_ticket.key(),
            expected_pda,
            LotteryError::WrongWinnerAccount
        );

        let total          = round.total_prize;
        let winner_share   = total * WINNER_BPS   / BPS_DENOM;
        let treasury_share = total * TREASURY_BPS / BPS_DENOM + round.slash_pool;
        let prize_bump     = round.prize_vault_bump;

        round.winner              = ctx.accounts.winner_ticket.owner;
        round.winner_share        = winner_share;
        round.treasury_share      = treasury_share;
        round.is_finalized        = true;
        round.slot_finalized      = clock.slot;
        round.rewards_distributed = true;
        round.staking_finalized   = true;
        round.vrf_timed_out       = true;

        // FIX 2: staking reward ratio (mismo cálculo que distribute_rewards)
        let total_staked = round.total_staked;
        if total_staked > 0 {
            let staking_reward_tokens = total * STAKING_BPS / BPS_DENOM;
            round.staking_reward_per_token_scaled =
                (staking_reward_tokens as u128)
                    .checked_mul(1_000_000_000u128).unwrap()
                    .checked_div(total_staked as u128).unwrap() as u64;
        }

        let prize_seeds: &[&[u8]] = &[
            b"prize_vault",
            &round_id.to_le_bytes(),
            &[prize_bump],
        ];
        let signer = &[prize_seeds];

        anchor_lang::system_program::transfer(
            CpiContext::new_with_signer(
                ctx.accounts.system_program.to_account_info(),
                anchor_lang::system_program::Transfer {
                    from: ctx.accounts.prize_vault.to_account_info(),
                    to:   ctx.accounts.winner_ticket_owner.to_account_info(),
                },
                signer,
            ),
            winner_share,
        )?;
        anchor_lang::system_program::transfer(
            CpiContext::new_with_signer(
                ctx.accounts.system_program.to_account_info(),
                anchor_lang::system_program::Transfer {
                    from: ctx.accounts.prize_vault.to_account_info(),
                    to:   ctx.accounts.treasury_vault.to_account_info(),
                },
                signer,
            ),
            treasury_share,
        )?;

        emit!(EmergencyFinalized {
            round_id,
            winner:       round.winner,
            winner_index,
            vrf_was_used: false,
        });
        Ok(())
    }
}

// ════════════════════════════════════════════════════════════════════════════

#[account]
#[derive(InitSpace)]
pub struct LotteryGlobal {
    pub admin:               Pubkey,
    pub round_count:         u64,
    pub max_token_supply:    u64,
    pub total_tokens_minted: u64,
    pub lottosol_mint:       Pubkey,
    pub treasury_vault:      Pubkey,
}

#[account]
#[derive(InitSpace)]
pub struct Round {
    pub id:           u64,
    pub ticket_price: u64,
    pub start_time:   i64,
    pub end_time:     i64,
    pub max_tickets:  u64,
    pub player_count: u64,
    pub winner:       Pubkey,
    pub is_finalized: bool,
    pub total_prize:  u64,
    pub admin:        Pubkey,
    pub bump:         u8,

    // — Staking (FIX 2: puramente SPL, no hay staking_pool en SOL) ——————
    pub total_staked:                   u64,
    pub staking_finalized:              bool,
    pub staking_token_vault:            Pubkey,
    pub staking_reward_per_token_scaled: u64,   // fixed-point ×1e9

    // — Reveal (FIX 4: solo lo necesario) ————————————————————————————————
    pub reveal_deadline:     i64,
    pub revealed_count:      u64,
    pub reveal_accumulator:  [u8; 32],   // para FIX 3 (emergency fallback)

    // — Slash (FIX 1: automático via batch) ——————————————————————————————
    pub reveal_deposit:   u64,
    pub slash_pool:       u64,
    pub slash_caller_bps: u16,
    pub slash_processed:  u64,   // FIX 6: cuántos tickets han sido evaluados

    // — VRF ——————————————————————————————————————————————————————————————
    pub vrf_account:          Pubkey,
    pub vrf_request_pending:  bool,
    pub vrf_result:           [u8; 32],
    pub vrf_timed_out:        bool,

    // — Vaults (FIX 5: prize_vault_bump para CPI signing) ————————————————
    pub prize_vault:      Pubkey,
    pub prize_vault_bump: u8,
    // FIX 2: staking_vault y staking_vault_bump eliminados

    // — Distribución ——————————————————————————————————————————————————————
    pub winner_share:        u64,
    pub treasury_share:      u64,
    pub rewards_distributed: bool,

    // — Auditoría —————————————————————————————————————————————————————————
    pub slot_finalized: u64,
}

#[account]
#[derive(InitSpace)]
pub struct Ticket {
    pub owner:         Pubkey,
    pub round_id:      u64,
    pub ticket_number: u64,
    pub commitment:    [u8; 32],
    pub revealed:      bool,
    pub slashed:       bool,
    pub bump:          u8,
}

#[account]
#[derive(InitSpace)]
pub struct TicketCounter {
    pub round_id: u64,
    pub count:    u64,
}

#[account]
#[derive(InitSpace)]
pub struct StakeEntry {
    pub user:     Pubkey,
    pub round_id: u64,
    pub amount:   u64,
    pub claimed:  bool,
    pub slashed:  bool,   // FIX 1: true si ticket del staker fue slasheado
    pub bump:     u8,
}

// ════════════════════════════════════════════════════════════════════════════
//  EVENTOS
// ════════════════════════════════════════════════════════════════════════════

#[event] pub struct RoundCreated         { pub round_id: u64 }
#[event] pub struct TicketPurchased      { pub round_id: u64, pub buyer: Pubkey, pub ticket_number: u64 }
#[event] pub struct SecretRevealed       { pub round_id: u64, pub ticket_number: u64 }
#[event] pub struct VrfRequested         { pub round_id: u64, pub vrf_account: Pubkey }
#[event] pub struct RoundFinalized       { pub round_id: u64, pub winner: Pubkey, pub winner_index: u64 }
#[event] pub struct RewardsDistributed   { pub round_id: u64, pub winner_share: u64, pub treasury_share: u64, pub slash_pool: u64 }
#[event] pub struct Staked               { pub round_id: u64, pub user: Pubkey, pub amount: u64 }
#[event] pub struct StakingRewardClaimed { pub round_id: u64, pub user: Pubkey, pub reward: u64 }
#[event] pub struct TicketSlashed        { pub round_id: u64, pub ticket_number: u64, pub slash_amount: u64 }
#[event] pub struct EmergencyFinalized   { pub round_id: u64, pub winner: Pubkey, pub winner_index: u64, pub vrf_was_used: bool }

// ════════════════════════════════════════════════════════════════════════════
//  ERRORES
// ════════════════════════════════════════════════════════════════════════════

#[error_code]
pub enum LotteryError {
    #[msg("No autorizado")]                                       Unauthorized,
    #[msg("Ronda ya finalizada")]                                 RoundAlreadyFinalized,
    #[msg("Ronda no activa")]                                     RoundNotActive,
    #[msg("Ronda llena")]                                         RoundFull,
    #[msg("La ronda no ha terminado aún")]                        RoundNotEnded,
    #[msg("No hay tickets")]                                      NoTickets,
    #[msg("Ronda no finalizada")]                                 RoundNotFinalized,
    #[msg("Ganador incorrecto")]                                  WrongWinnerAccount,
    #[msg("Ticket incorrecto en batch")]                          WrongTicketAccount,
    #[msg("Staking ya finalizado")]                               StakingAlreadyFinalized,
    #[msg("La ronda ya terminó")]                                 RoundEnded,
    #[msg("Staking no finalizado")]                               StakingNotFinalized,
    #[msg("Recompensas no distribuidas")]                         RewardsNotDistributed,
    #[msg("Recompensas ya distribuidas")]                         RewardsAlreadyDistributed,
    #[msg("Ya reclamaste tu recompensa")]                         AlreadyClaimed,
    #[msg("No hay stakers")]                                      NoStakers,
    #[msg("Sin recompensa")]                                      NoReward,
    #[msg("Fase de compra terminada")]                            CommitPhaseEnded,
    #[msg("Fase de revelación no activa")]                        RevealPhaseNotActive,
    #[msg("Fase de revelación aún activa")]                       RevealPhaseStillActive,
    #[msg("Secreto inválido")]                                    InvalidSecret,
    #[msg("Ya revelaste tu secreto")]                             AlreadyRevealed,
    #[msg("Suministro máximo de tokens excedido")]                TokenSupplyExceeded,
    #[msg("Balance insuficiente")]                                InsufficientBalance,
    #[msg("Vault de staking inválido")]                           InvalidStakingVault,
    #[msg("Mint inválida")]                                       InvalidMint,
    #[msg("VRF ya solicitado")]                                   VrfAlreadyRequested,
    #[msg("VRF no solicitado — llama request_vrf primero")]       VrfNotRequested,
    #[msg("VRF inválido o cuenta incorrecta")]                    VrfInvalid,
    #[msg("VRF aún no tiene resultado — espera al oracle")]       VrfNotReady,
    #[msg("Demasiado pronto para emergencia")]                    EmergencyTooEarly,
    #[msg("Parámetro inválido")]                                  InvalidParam,
    #[msg("Ticket ya fue slasheado")]                             TicketAlreadySlashed,
    #[msg("Stake slasheado — no puedes reclamar reward")]         StakeSlashed,
    #[msg("No hay depósito de reveal para slashear")]             NoSlashDeposit,
    #[msg("Slash batch pendiente — procesa tickets primero")]     SlashBatchPending,
    #[msg("Slash ya completado para esta ronda")]                 SlashAlreadyComplete,
}
