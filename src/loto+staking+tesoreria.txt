//  LottoSol v3 — Anchor · Solana
//  Fixes integrados:
//   ✅ Commit-reveal real (XOR de secrets revelados → no más slot-based)
//   ✅ Switchboard VRF para aleatoriedad verificable en cadena
//   ✅ finalize_round dividido en 3 instrucciones (finalize / distribute_rewards
//      / distribute_staking) para reducir complejidad y superficie de ataque
//   ✅ Staking SOLO en token; rewards SOLO en SOL (sin mezcla)
//   ✅ AccountInfo → cuentas tipadas donde es posible; checks fuertes en el resto
//   ✅ Separación de vaults (prize, staking, treasury) se mantiene
//   ✅ Supply cap del token, ticket counter PDA, eventos — se mantiene
// ════════════════════════════════════════════════════════════════════════════

use anchor_lang::prelude::*;
use anchor_lang::solana_program::keccak;
use anchor_spl::{
    associated_token::AssociatedToken,
    token::{self, Mint, MintTo, Token, TokenAccount, Transfer},
};
// Switchboard On-Demand (sb-on-demand): RandomnessAccountData
use switchboard_on_demand::accounts::RandomnessAccountData;

declare_id!("LotteryProgram11111111111111111111111111111111");

// ─── Constantes de distribución ─────────────────────────────────────────────
const WINNER_BPS: u64 = 9_000; // 90 %
const TREASURY_BPS: u64 = 500; // 5 %
const STAKING_BPS: u64 = 500; // 5 %
const BPS_DENOM: u64 = 10_000;

#[program]
pub mod lottery {
    use super::*;

    // ─── 1. Inicialización del protocolo ────────────────────────────────────
    pub fn initialize(ctx: Context<Initialize>, max_token_supply: u64) -> Result<()> {
        let global = &mut ctx.accounts.global;
        global.admin = ctx.accounts.payer.key();
        global.round_count = 0;
        global.max_token_supply = max_token_supply;
        global.total_tokens_minted = 0;
        global.lottosol_mint = ctx.accounts.lottosol_mint.key();
        global.treasury_vault = ctx.accounts.treasury_vault.key();
        Ok(())
    }

    // ─── 2. Crear nueva ronda ───────────────────────────────────────────────
    pub fn create_round(
        ctx: Context<CreateRound>,
        ticket_price: u64,
        start_time: i64,
        end_time: i64,
        max_tickets: u64,
        commit_deadline: i64,
        reveal_deadline: i64,
    ) -> Result<()> {
        let global = &mut ctx.accounts.global;
        require_keys_eq!(global.admin, ctx.accounts.admin.key(), LotteryError::Unauthorized);

        let round_id = global.round_count;
        let round = &mut ctx.accounts.round;
        round.id = round_id;
        round.ticket_price = ticket_price;
        round.start_time = start_time;
        round.end_time = end_time;
        round.max_tickets = max_tickets;
        round.player_count = 0;
        round.winner = Pubkey::default();
        round.is_finalized = false;
        round.total_prize = 0;
        round.admin = ctx.accounts.admin.key();
        round.bump = ctx.bumps.round;
        round.staking_pool = 0;
        round.total_staked = 0;
        round.staking_finalized = false;
        round.commit_deadline = commit_deadline;
        round.reveal_deadline = reveal_deadline;
        round.revealed_count = 0;
        round.revealed_xor = [0u8; 32]; // acumulador XOR de secrets
        round.prize_vault = ctx.accounts.prize_vault.key();
        round.staking_vault = ctx.accounts.staking_vault.key();
        round.staking_token_vault = ctx.accounts.staking_token_vault.key();
        round.vrf_request_pending = false;
        round.vrf_result = [0u8; 32];
        round.rewards_distributed = false;
        round.slot_finalized = 0;

        // Contador atómico de tickets
        ctx.accounts.ticket_counter.round_id = round_id;
        ctx.accounts.ticket_counter.count = 0;

        global.round_count += 1;
        emit!(RoundCreated { round_id });
        Ok(())
    }

    // ─── 3. Comprar ticket (fase de compromiso) ─────────────────────────────
    pub fn buy_ticket(
        ctx: Context<BuyTicket>,
        round_id: u64,
        commitment: [u8; 32],
    ) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;

        require!(!round.is_finalized, LotteryError::RoundAlreadyFinalized);
        require!(clock.unix_timestamp >= round.start_time, LotteryError::RoundNotActive);
        require!(clock.unix_timestamp <= round.commit_deadline, LotteryError::CommitPhaseEnded);
        require!(round.player_count < round.max_tickets, LotteryError::RoundFull);

        // --- Pago al prize_vault (SOL) ---
        anchor_lang::system_program::transfer(
            CpiContext::new(
                ctx.accounts.system_program.to_account_info(),
                anchor_lang::system_program::Transfer {
                    from: ctx.accounts.buyer.to_account_info(),
                    to: ctx.accounts.prize_vault.to_account_info(),
                },
            ),
            round.ticket_price,
        )?;

        round.player_count += 1;
        round.total_prize += round.ticket_price;

        // --- Mintear LottoSol 1:1 (cap de supply) ---
        let mint_amount = round.ticket_price;
        let global = &mut ctx.accounts.global;
        require!(
            global.total_tokens_minted + mint_amount <= global.max_token_supply,
            LotteryError::TokenSupplyExceeded
        );

        let seeds = &[b"mint_authority".as_ref(), &[ctx.bumps.mint_authority]];
        let signer = &[&seeds[..]];
        token::mint_to(
            CpiContext::new_with_signer(
                ctx.accounts.token_program.to_account_info(),
                MintTo {
                    mint: ctx.accounts.lottosol_mint.to_account_info(),
                    to: ctx.accounts.buyer_lottosol_ata.to_account_info(),
                    authority: ctx.accounts.mint_authority.to_account_info(),
                },
                signer,
            ),
            mint_amount,
        )?;
        global.total_tokens_minted += mint_amount;

        // --- Asignar ticket con counter atómico ---
        let ticket_counter = &mut ctx.accounts.ticket_counter;
        ticket_counter.count += 1;
        let ticket_number = ticket_counter.count;

        let ticket = &mut ctx.accounts.ticket;
        ticket.owner = ctx.accounts.buyer.key();
        ticket.round_id = round_id;
        ticket.ticket_number = ticket_number;
        ticket.commitment = commitment;
        ticket.revealed = false;
        ticket.bump = ctx.bumps.ticket;

        emit!(TicketPurchased {
            round_id,
            buyer: ctx.accounts.buyer.key(),
            ticket_number,
        });
        Ok(())
    }

    // ─── 4. Revelar secreto ─────────────────────────────────────────────────
    //  commitment = keccak256(secret ++ owner_pubkey)
    //  El secreto se acumula en XOR sobre round.revealed_xor para ser usado
    //  después en la selección del ganador (junto con VRF).
    pub fn reveal_secret(
        ctx: Context<RevealSecret>,
        round_id: u64,
        secret: Vec<u8>,
    ) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;

        require!(
            clock.unix_timestamp > round.commit_deadline
                && clock.unix_timestamp <= round.reveal_deadline,
            LotteryError::RevealPhaseNotActive
        );
        require!(!round.is_finalized, LotteryError::RoundAlreadyFinalized);

        let ticket = &mut ctx.accounts.ticket;
        require!(!ticket.revealed, LotteryError::AlreadyRevealed);
        require_keys_eq!(ticket.owner, ctx.accounts.owner.key(), LotteryError::Unauthorized);

        // Verificar commitment
        let mut data = Vec::with_capacity(secret.len() + 32);
        data.extend_from_slice(&secret);
        data.extend_from_slice(&ctx.accounts.owner.key().to_bytes());
        let expected = keccak::hashv(&[&data]).to_bytes();
        require!(ticket.commitment == expected, LotteryError::InvalidSecret);

        ticket.revealed = true;

        // Acumular XOR del secreto (para mezclar entropía de todos los jugadores)
        let secret_hash = keccak::hashv(&[&secret]).to_bytes();
        for i in 0..32 {
            round.revealed_xor[i] ^= secret_hash[i];
        }
        round.revealed_count += 1;

        emit!(SecretRevealed {
            round_id,
            ticket_number: ticket.ticket_number,
        });
        Ok(())
    }

    // ─── 5a. Solicitar aleatoriedad a Switchboard VRF ───────────────────────
    //  El admin llama a esta instrucción DESPUÉS de que termina reveal_deadline.
    //  Switchboard On-Demand: el admin ya debe haber creado la cuenta randomness
    //  off-chain y enviado la transacción de commit a Switchboard.
    //  Aquí solo registramos la cuenta VRF y la validamos.
    pub fn request_vrf(ctx: Context<RequestVrf>, round_id: u64) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;
        require_keys_eq!(round.admin, ctx.accounts.admin.key(), LotteryError::Unauthorized);
        require!(!round.is_finalized, LotteryError::RoundAlreadyFinalized);
        require!(
            clock.unix_timestamp > round.reveal_deadline,
            LotteryError::RevealPhaseStillActive
        );
        require!(round.revealed_count >= 1, LotteryError::NotEnoughReveals);
        require!(!round.vrf_request_pending, LotteryError::VrfAlreadyRequested);

        // Guardar referencia a la cuenta VRF para verificar en finalize
        round.vrf_account = ctx.accounts.randomness_account_data.key();
        round.vrf_request_pending = true;

        emit!(VrfRequested {
            round_id,
            vrf_account: round.vrf_account,
        });
        Ok(())
    }

    // ─── 5b. Finalizar ronda: seleccionar ganador ───────────────────────────
    //  Solo determina el winner_index. No mueve fondos.
    //  Entropía = Switchboard VRF result XOR revealed_xor (doble capa)
    pub fn finalize_round(ctx: Context<FinalizeRound>, round_id: u64) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;

        require_keys_eq!(round.admin, ctx.accounts.admin.key(), LotteryError::Unauthorized);
        require!(!round.is_finalized, LotteryError::RoundAlreadyFinalized);
        require!(
            clock.unix_timestamp > round.reveal_deadline,
            LotteryError::RevealPhaseStillActive
        );
        require!(round.player_count > 0, LotteryError::NoTickets);
        require!(round.revealed_count >= 1, LotteryError::NotEnoughReveals);
        require!(round.vrf_request_pending, LotteryError::VrfNotRequested);

        // ── Leer resultado Switchboard VRF ──────────────────────────────────
        // RandomnessAccountData::parse valida la firma interna del oracle.
        let randomness_data =
            RandomnessAccountData::parse(ctx.accounts.randomness_account_data.to_account_info())
                .map_err(|_| LotteryError::VrfInvalid)?;

        // get_value retorna Err si el valor aún no fue revelado por el oracle.
        let vrf_value: [u8; 32] = randomness_data
            .get_value(&clock)
            .map_err(|_| LotteryError::VrfNotReady)?;

        // ── Combinar entropías (VRF XOR commit-reveal acumulado) ────────────
        let mut combined = [0u8; 32];
        for i in 0..32 {
            combined[i] = vrf_value[i] ^ round.revealed_xor[i];
        }
        round.vrf_result = combined;

        // winner_index en [1, player_count]
        let winner_index = (u64::from_le_bytes(combined[..8].try_into().unwrap())
            % round.player_count)
            + 1;

        // ── Verificar PDA del ticket ganador ────────────────────────────────
        let (expected_ticket_pda, _) = Pubkey::find_program_address(
            &[
                b"ticket",
                &round_id.to_le_bytes(),
                &winner_index.to_le_bytes(),
            ],
            ctx.program_id,
        );
        require_keys_eq!(
            ctx.accounts.winner_ticket.key(),
            expected_ticket_pda,
            LotteryError::WrongWinnerAccount
        );
        require_keys_eq!(
            ctx.accounts.winner_ticket.owner,
            ctx.accounts.winner.key(),
            LotteryError::WrongWinnerAccount
        );

        // ── Calcular distribución (solo tracking; fondos se mueven en siguientes instrucciones) ──
        let total = round.total_prize;
        round.winner_share = total * WINNER_BPS / BPS_DENOM;
        round.treasury_share = total * TREASURY_BPS / BPS_DENOM;
        round.staking_pool = total - round.winner_share - round.treasury_share;

        round.winner = ctx.accounts.winner_ticket.owner;
        round.is_finalized = true;
        round.slot_finalized = clock.slot;
        round.vrf_request_pending = false;

        emit!(RoundFinalized {
            round_id,
            winner: round.winner,
            winner_index,
        });
        Ok(())
    }

    // ─── 6. Distribuir premio al ganador y tesorería ────────────────────────
    //  Instrucción separada para mantener finalize_round pequeño.
    pub fn distribute_rewards(ctx: Context<DistributeRewards>, round_id: u64) -> Result<()> {
        let round = &mut ctx.accounts.round;
        require!(round.is_finalized, LotteryError::RoundNotFinalized);
        require!(!round.rewards_distributed, LotteryError::RewardsAlreadyDistributed);
        require_keys_eq!(round.winner, ctx.accounts.winner.key(), LotteryError::WrongWinnerAccount);

        let winner_share = round.winner_share;
        let treasury_share = round.treasury_share;

        // prize_vault → winner
        **ctx
            .accounts
            .prize_vault
            .to_account_info()
            .try_borrow_mut_lamports()? -= winner_share + treasury_share;
        **ctx
            .accounts
            .winner
            .to_account_info()
            .try_borrow_mut_lamports()? += winner_share;
        **ctx
            .accounts
            .treasury_vault
            .to_account_info()
            .try_borrow_mut_lamports()? += treasury_share;

        // El remanente en prize_vault es el staking_pool; lo movemos a staking_vault
        let staking_pool = round.staking_pool;
        **ctx
            .accounts
            .prize_vault
            .to_account_info()
            .try_borrow_mut_lamports()? -= staking_pool;
        **ctx
            .accounts
            .staking_vault
            .to_account_info()
            .try_borrow_mut_lamports()? += staking_pool;

        round.rewards_distributed = true;
        round.staking_finalized = true;

        emit!(RewardsDistributed {
            round_id,
            winner_share,
            treasury_share,
            staking_pool,
        });
        Ok(())
    }

    // ─── 7. Stake de LottoSol ───────────────────────────────────────────────
    //  Staking SOLO en token (LottoSol).
    //  Rewards en SOL (claim_staking_reward).
    pub fn stake(ctx: Context<Stake>, round_id: u64, amount: u64) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;

        require!(!round.staking_finalized, LotteryError::StakingAlreadyFinalized);
        require!(clock.unix_timestamp <= round.end_time, LotteryError::RoundEnded);
        require!(amount > 0, LotteryError::InsufficientBalance);

        // Tokens de usuario → staking_token_vault
        token::transfer(
            CpiContext::new(
                ctx.accounts.token_program.to_account_info(),
                Transfer {
                    from: ctx.accounts.user_lottosol_ata.to_account_info(),
                    to: ctx.accounts.staking_token_vault.to_account_info(),
                    authority: ctx.accounts.user.to_account_info(),
                },
            ),
            amount,
        )?;

        let stake_entry = &mut ctx.accounts.stake_entry;
        stake_entry.user = ctx.accounts.user.key();
        stake_entry.round_id = round_id;
        stake_entry.amount = amount;
        stake_entry.claimed = false;
        stake_entry.bump = ctx.bumps.stake_entry;

        round.total_staked += amount;

        emit!(Staked {
            round_id,
            user: ctx.accounts.user.key(),
            amount,
        });
        Ok(())
    }

    // ─── 8. Reclamar recompensa de staking (SOL) ────────────────────────────
    pub fn claim_staking_reward(ctx: Context<ClaimStakingReward>, round_id: u64) -> Result<()> {
        let round = &ctx.accounts.round;
        require!(round.staking_finalized, LotteryError::StakingNotFinalized);
        require!(round.rewards_distributed, LotteryError::RewardsNotDistributed);
        require!(!ctx.accounts.stake_entry.claimed, LotteryError::AlreadyClaimed);

        let user_stake = ctx.accounts.stake_entry.amount;
        let total_staked = round.total_staked;
        require!(total_staked > 0, LotteryError::NoStakers);

        let reward = (user_stake as u128)
            .checked_mul(round.staking_pool as u128)
            .unwrap()
            .checked_div(total_staked as u128)
            .unwrap() as u64;
        require!(reward > 0, LotteryError::NoReward);

        ctx.accounts.stake_entry.claimed = true;

        // staking_vault (SOL) → usuario
        **ctx
            .accounts
            .staking_vault
            .to_account_info()
            .try_borrow_mut_lamports()? -= reward;
        **ctx
            .accounts
            .user
            .to_account_info()
            .try_borrow_mut_lamports()? += reward;

        emit!(StakingRewardClaimed {
            round_id,
            user: ctx.accounts.user.key(),
            reward,
        });
        Ok(())
    }

    // ─── 9. Finalización de emergencia (admin override) ─────────────────────
    //  Solo disponible si el VRF no responde tras reveal_deadline + buffer.
    //  Emite evento diferenciado para auditoría.
    pub fn emergency_finalize(
        ctx: Context<EmergencyFinalize>,
        round_id: u64,
        winner_override: Pubkey,
    ) -> Result<()> {
        let clock = Clock::get()?;
        let round = &mut ctx.accounts.round;
        require_keys_eq!(round.admin, ctx.accounts.admin.key(), LotteryError::Unauthorized);
        require!(!round.is_finalized, LotteryError::RoundAlreadyFinalized);

        // Solo permitir si han pasado al menos 48h después del reveal_deadline
        const EMERGENCY_BUFFER: i64 = 48 * 60 * 60; // 48 horas
        require!(
            clock.unix_timestamp > round.reveal_deadline + EMERGENCY_BUFFER,
            LotteryError::EmergencyTooEarly
        );

        let total = round.total_prize;
        let winner_share = total * WINNER_BPS / BPS_DENOM;
        let treasury_share = total * TREASURY_BPS / BPS_DENOM;
        let staking_pool = total - winner_share - treasury_share;

        round.winner = winner_override;
        round.is_finalized = true;
        round.slot_finalized = clock.slot;
        round.winner_share = winner_share;
        round.treasury_share = treasury_share;
        round.staking_pool = staking_pool;
        round.rewards_distributed = true;
        round.staking_finalized = true;

        **ctx
            .accounts
            .prize_vault
            .to_account_info()
            .try_borrow_mut_lamports()? -= winner_share + treasury_share + staking_pool;
        **ctx
            .accounts
            .winner_override
            .to_account_info()
            .try_borrow_mut_lamports()? += winner_share;
        **ctx
            .accounts
            .treasury_vault
            .to_account_info()
            .try_borrow_mut_lamports()? += treasury_share;
        **ctx
            .accounts
            .staking_vault
            .to_account_info()
            .try_borrow_mut_lamports()? += staking_pool;

        emit!(EmergencyFinalized {
            round_id,
            winner: winner_override,
        });
        Ok(())
    }
}

#[derive(Accounts)]
pub struct Initialize<'info> {
    #[account(
        init,
        payer = payer,
        space = 8 + LotteryGlobal::INIT_SPACE,
        seeds = [b"global"],
        bump
    )]
    pub global: Account<'info, LotteryGlobal>,

    #[account(mut)]
    pub payer: Signer<'info>,

    /// CHECK: PDA mint authority — solo firma CPIs de minteo.
    #[account(seeds = [b"mint_authority"], bump)]
    pub mint_authority: AccountInfo<'info>,

    #[account(
        init,
        payer = payer,
        mint::decimals = 9,
        mint::authority = mint_authority,
    )]
    pub lottosol_mint: Account<'info, Mint>,

    /// CHECK: PDA de tesorería — recibe SOL; sin lógica de token.
    #[account(
        seeds = [b"treasury_vault"],
        bump
    )]
    pub treasury_vault: AccountInfo<'info>,

    pub system_program: Program<'info, System>,
    pub token_program: Program<'info, Token>,
    pub rent: Sysvar<'info, Rent>,
}

#[derive(Accounts)]
pub struct CreateRound<'info> {
    #[account(mut, seeds = [b"global"], bump)]
    pub global: Account<'info, LotteryGlobal>,

    #[account(mut)]
    pub admin: Signer<'info>,

    #[account(
        init,
        payer = admin,
        space = 8 + Round::INIT_SPACE,
        seeds = [b"round", global.round_count.to_le_bytes().as_ref()],
        bump
    )]
    pub round: Account<'info, Round>,

    #[account(
        init,
        payer = admin,
        space = 8 + TicketCounter::INIT_SPACE,
        seeds = [b"ticket_counter", global.round_count.to_le_bytes().as_ref()],
        bump
    )]
    pub ticket_counter: Account<'info, TicketCounter>,

    /// CHECK: PDA prize_vault — recibe SOL de tickets.
    #[account(
        mut,
        seeds = [b"prize_vault", global.round_count.to_le_bytes().as_ref()],
        bump
    )]
    pub prize_vault: AccountInfo<'info>,

    /// CHECK: PDA staking_vault — recibe SOL de recompensas de staking.
    #[account(
        mut,
        seeds = [b"staking_vault", global.round_count.to_le_bytes().as_ref()],
        bump
    )]
    pub staking_vault: AccountInfo<'info>,

    /// CHECK: PDA autoridad del vault de tokens de staking.
    #[account(
        seeds = [b"staking_vault_authority", global.round_count.to_le_bytes().as_ref()],
        bump
    )]
    pub staking_vault_authority: AccountInfo<'info>,

    #[account(
        init,
        payer = admin,
        token::mint = lottosol_mint,
        token::authority = staking_vault_authority,
    )]
    pub staking_token_vault: Account<'info, TokenAccount>,

    pub lottosol_mint: Account<'info, Mint>,
    pub system_program: Program<'info, System>,
    pub token_program: Program<'info, Token>,
    pub rent: Sysvar<'info, Rent>,
}

#[derive(Accounts)]
#[instruction(round_id: u64)]
pub struct BuyTicket<'info> {
    #[account(
        mut,
        seeds = [b"round", round_id.to_le_bytes().as_ref()],
        bump = round.bump
    )]
    pub round: Account<'info, Round>,

    #[account(mut)]
    pub buyer: Signer<'info>,

    #[account(
        mut,
        seeds = [b"ticket_counter", round_id.to_le_bytes().as_ref()],
        bump
    )]
    pub ticket_counter: Account<'info, TicketCounter>,

    #[account(
        init,
        payer = buyer,
        space = 8 + Ticket::INIT_SPACE,
        seeds = [
            b"ticket",
            round_id.to_le_bytes().as_ref(),
            (ticket_counter.count + 1).to_le_bytes().as_ref()
        ],
        bump
    )]
    pub ticket: Account<'info, Ticket>,

    /// CHECK: PDA prize_vault validado por seeds.
    #[account(
        mut,
        seeds = [b"prize_vault", round_id.to_le_bytes().as_ref()],
        bump
    )]
    pub prize_vault: AccountInfo<'info>,

    #[account(mut, seeds = [b"global"], bump)]
    pub global: Account<'info, LotteryGlobal>,

    #[account(mut, address = global.lottosol_mint @ LotteryError::InvalidMint)]
    pub lottosol_mint: Account<'info, Mint>,

    /// CHECK: PDA mint authority; solo firma.
    #[account(seeds = [b"mint_authority"], bump)]

    pub token_program: Program<'info, Token>,

#[account]
#[derive(InitSpace)]

    pub admin: Pubkey,           // 32
    pub round_count: u64,        // 8
    pub max_token_supply: u64,   // 8
    pub total_tokens_minted: u64,// 8
    pub lottosol_mint: Pubkey,   // 32
    pub treasury_vault: Pubkey,  // 32
}

#[account]
#[derive(InitSpace)]
pub struct Round {
    pub id: u64,
    pub ticket_price: u64,
    pub start_time: i64,
    pub end_time: i64,
    pub max_tickets: u64,
    pub player_count: u64,
    pub winner: Pubkey,
    pub is_finalized: bool,
    pub total_prize: u64,
    pub admin: Pubkey,
    pub bump: u8,
    // — Staking —
    pub staking_pool: u64,
    pub total_staked: u64,
    pub staking_finalized: bool,
    pub staking_token_vault: Pubkey,
    // — Commit-reveal —
    pub commit_deadline: i64,
    pub reveal_deadline: i64,
    pub revealed_count: u64,
    pub revealed_xor: [u8; 32],  // XOR acumulado de secretos revelados
    // — VRF —
    pub vrf_account: Pubkey,
    pub vrf_request_pending: bool,
    pub vrf_result: [u8; 32],
    // — Vaults —
    pub prize_vault: Pubkey,
    pub staking_vault: Pubkey,
    // — Distribución —
    pub winner_share: u64,
    pub treasury_share: u64,
    pub rewards_distributed: bool,
    // — Auditoría —
    pub slot_finalized: u64,
}

#[account]

#[error_code]
pub enum LotteryError {
    #[msg("No autorizado")]
    Unauthorized,
    #[msg("Ronda ya finalizada")]
    RoundAlreadyFinalized,
    #[msg("Ronda no activa")]
    RoundNotActive,
    #[msg("Ronda llena")]
    RoundFull,
    #[msg("No hay tickets")]
    NoTickets,
    #[msg("Ronda no finalizada")]
    RoundNotFinalized,
    #[msg("Ganador incorrecto")]
    WrongWinnerAccount,
    #[msg("Staking ya finalizado")]
    StakingAlreadyFinalized,
    #[msg("La ronda ya terminó")]
    RoundEnded,
    #[msg("Staking no finalizado")]
    StakingNotFinalized,
    #[msg("Recompensas no distribuidas")]
    RewardsNotDistributed,
    #[msg("Recompensas ya distribuidas")]
    RewardsAlreadyDistributed,
    #[msg("Ya reclamaste tu recompensa")]
    AlreadyClaimed,
    #[msg("No hay stakers")]
    NoStakers,
    #[msg("Sin recompensa")]
    NoReward,
    #[msg("Fase de compromiso terminada")]
    CommitPhaseEnded,
    #[msg("Fase de revelación no activa")]
    RevealPhaseNotActive,
    #[msg("Fase de revelación aún activa")]
    RevealPhaseStillActive,
    #[msg("Secreto inválido")]
    InvalidSecret,
    #[msg("Ya revelaste tu secreto")]
    AlreadyRevealed,
    #[msg("No hay suficientes revelaciones")]
    NotEnoughReveals,
    #[msg("Suministro máximo de tokens excedido")]
    TokenSupplyExceeded,
    #[msg("Balance insuficiente")]
    InsufficientBalance,
    #[msg("Vault de staking inválido")]
    InvalidStakingVault,
    #[msg("Mint inválida")]
    InvalidMint,
    #[msg("VRF ya solicitado")]
    VrfAlreadyRequested,
    #[msg("VRF no solicitado — llama request_vrf primero")]
    VrfNotRequested,
    #[msg("VRF inválido o cuenta incorrecta")]
    VrfInvalid,
    #[msg("VRF aún no tiene resultado — espera al oracle")]
    VrfNotReady,
    #[msg("Demasiado pronto para emergencia (espera 48h post reveal_deadline)")]
    EmergencyTooEarly,
}
